    43→def start_mcp_server_background(host="0.0.0.0", port=8768, transport="sse"):
    44→    """在后台线程启动 MCP 服务器"""
    45→    global _mcp_thread, _mcp_running
    46→    if _mcp_running:
    47→        return "MCP 服务器已在运行中"
    48→
    49→    def _run():
    50→        global _mcp_running
    51→        try:
    52→            from fridamcp.server import create_mcp_server, run_sse_server, run_streamable_http_server
    53→            mcp = create_mcp_server()
    54→            _mcp_running = True
    55→            logger.info(f"MCP server starting on {host}:{port} ({transport})")
    56→            if transport == "sse":
    57→                asyncio.run(run_sse_server(mcp, host, port))
    58→            elif transport == "http":
    59→                asyncio.run(run_streamable_http_server(mcp, host, port))
    60→        except Exception as e:
    61→            logger.error(f"MCP server error: {e}")
    62→        finally:
    63→            _mcp_running = False
    64→
    65→    _mcp_thread = threading.Thread(target=_run, daemon=True)
    66→    _mcp_thread.start()
    67→    time.sleep(1)
    68→    return f"MCP 服务器已启动: {host}:{port} ({transport})"
    69→

    71→def stop_mcp_server():
    72→    """停止 MCP 服务器"""
    73→    global _mcp_running
    74→    if not _mcp_running:
    75→        return "MCP 服务器未运行"
    76→    _mcp_running = False
    77→    session_manager.close_all()

   233→def hook_java_method(session_id, class_name, method_name):
   234→    """Hook Java 方法"""
   235→    try:
   236→        import uuid
   237→        from fridamcp.modules.hook import HOOK_JAVA_METHOD_TEMPLATE
   238→        hook_id = f"hook_{uuid.uuid4().hex[:8]}"
   239→        source = HOOK_JAVA_METHOD_TEMPLATE % {
   240→            "hook_id": hook_id,
   241→            "class_name": class_name,
   242→            "method_name": method_name,
   243→        }
   244→        result = frida_client.execute_script(session_id, source, script_name=hook_id)
   245→        session = session_manager.get_session(session_id)
   246→        if session:
   247→            session.add_hook(hook_id, {
   248→                "type": "java_method",
   249→                "class_name": class_name,
   250→                "method_name": method_name,
   251→                "script_id": result["script_id"],
   252→            })
   253→        return f"✅ Hook 已安装\nHook ID: {hook_id}\nScript ID: {result['script_id']}"
   254→    except Exception as e:
   255→        return f"❌ Hook 失败: {e}"

   286→def get_hook_messages(session_id, clear=False):

   289→        messages = frida_client.get_messages(session_id, clear=clear)

   293→        for m in messages[-50:]:  # 最近50条

   295→            mtype = msg.get("type", "unknown")
   296→            if mtype == "hook_call":
   297→                lines.append(f"[CALL] {msg.get('className','')}.{msg.get('methodName','')} args={msg.get('args',[])}")

   300→            elif mtype == "hook_attached":
   301→                lines.append(f"[INFO] Hook attached: {msg.get('hookId','')}")
   302→            elif mtype == "error":
   303→                lines.append(f"[ERR]  {msg.get('message','')}")
   304→            else:
   305→                lines.append(f"[{mtype}] {json.dumps(msg, ensure_ascii=False)[:200]}")

   327→def unhook(session_id, hook_id):
   328→    """移除 Hook"""
   329→    try:
   330→        session = session_manager.get_session(session_id)
   331→        if session is None:
   332→            return "会话不存在"
   333→        if not hook_id:
   334→            count = len(session.hooks)
   335→            session.unload_all_scripts()
   336→            session.hooks.clear()
   337→            return f"✅ 已移除所有 Hook（{count} 个）"
   338→        info = session.hooks.get(hook_id)
   339→        if info:
   340→            session.unload_script(info["script_id"])
   341→            session.remove_hook(hook_id)
   342→            return f"✅ 已移除 Hook: {hook_id}"
   343→        return f"Hook 不存在: {hook_id}"
   344→    except Exception as e:
   345→        return f"❌ 失败: {e}"

   363→def search_memory(session_id, pattern, max_results=100):
   364→    """搜索内存"""
   365→    try:
   366→        from fridamcp.modules.memory import MEMORY_SEARCH_TEMPLATE
   367→        import uuid
   368→        clean = pattern.replace(" ", "").replace("\\x", "")
   369→        is_hex = all(c in "0123456789abcdefABCDEF" for c in clean) and len(clean) % 2 == 0
   370→        if not is_hex:
   371→            hex_pattern = " ".join(f"{ord(c):02x}" for c in pattern)
   372→        else:
   373→            hex_pattern = " ".join(clean[i:i+2] for i in range(0, len(clean), 2))
   374→
   375→        script_id = f"search_{uuid.uuid4().hex[:8]}"
   376→        result = frida_client.execute_script(session_id, MEMORY_SEARCH_TEMPLATE, script_name=script_id)
   377→        ret = frida_client.call_script_function(session_id, result["script_id"], "search", [hex_pattern, int(max_results)])
   378→        return json.dumps(ret, indent=2, ensure_ascii=False)
   379→    except Exception as e:
   380→        return f"错误: {e}"

   436→    try:
   437→        from fridamcp.modules.network import _capture_buffer

   439→        if clear:
   440→            _capture_buffer.clear()
   441→        if not items:
   442→            return "无捕获数据"
   443→        lines = []
   444→        for item in items[-100:]:
   445→            t = item.get("type", "")
   446→            data = item.get("data", "")
   447→            if len(data) > 200:
   448→                data = data[:200] + "..."
   449→            ip = item.get("ip", "")
   450→            port = item.get("port", "")
   451→            size = item.get("size", "")
   452→            if t == "ssl_write":
   453→                lines.append(f"[SSL→] {size}B: {data}")
   454→            elif t == "ssl_read":
   455→                lines.append(f"[SSL←] {size}B: {data}")
   456→            elif t == "socket_connect":
   457→                lines.append(f"[CONN] {ip}:{port}")
   458→            elif t == "socket_send":
   459→                lines.append(f"[SEND] {size}B: {data}")
   460→            else:
   461→                lines.append(f"[{t}] {data}")
   462→        return "\n".join(lines)
   463→    except Exception as e:
   464→        return f"错误: {e}"

   471→def get_server_logs(max_entries=100):
   472→    """获取服务器日志"""
   473→    try:
   474→        buf = get_log_buffer()
   475→        entries = list(buf)[-max_entries:]
   476→        lines = []
   477→        for e in entries:
   478→            time_obj = e.get("time")
   479→            level_obj = e.get("level")
   480→            time_str = time_obj.isoformat() if hasattr(time_obj, "isoformat") else str(time_obj)
   481→            level_str = level_obj.name if hasattr(level_obj, "name") else str(level_obj)
   482→            msg = e.get("message", "")
   483→            lines.append(f"[{time_str}] {level_str}: {msg}")
   484→        return "\n".join(lines) if lines else "无日志"
   485→    except Exception as e:
   486→        return f"错误: {e}"

   514→def inject_apk(input_apk, output_apk, arch, use_apktool, application_class):
   515→    """注入 APK"""
   516→    try:
   517→        from fridamcp.utils.apk_injector import inject_gadget, inject_with_apktool
   518→        if not input_apk:
   519→            return "❌ 请选择输入 APK"
   520→        if not output_apk:
   521→            output_apk = input_apk.replace(".apk", "_injected.apk")
   522→
   523→        if use_apktool:
   524→            result = inject_with_apktool(input_apk, output_apk, arch=arch or None, application_class=application_class or None)
   525→        else:
   526→            result = inject_gadget(input_apk, output_apk, arch=arch or None)
   527→
   528→        if result.get("success"):
   529→            return f"✅ 注入成功!\n输出: {result.get('output_apk', output_apk)}\n架构: {result.get('archs', [])}"
   530→        else:
   531→            return f"❌ 注入失败: {result.get('error', '未知错误')}\n{result.get('detail','')}\n{result.get('note','')}"
   532→    except Exception as e:

   861→        description="FridaMCP GUI - Android Frida 动态分析平台"

   870→    )
   871→    parser.add_argument(
   872→        "--mcp", action="store_true",
   873→        help="同时启动 MCP 服务器（端口 8768）",
   874→    )
   875→    parser.add_argument(
   876→        "--mcp-port", type=int, default=8768,
   877→        help="MCP 服务器端口（默认 8768）",
   878→    )
   879→    parser.add_argument(
   880→        "--device-type", default="usb",
   881→        choices=["usb", "remote", "local"],
   882→        help="Frida 设备类型（默认 usb）",
   883→    )
   884→
   885→    args = parser.parse_args()
   886→
   887→    # 设置环境变量
   888→    os.environ["FRIDA_DEVICE_TYPE"] = args.device_type
   889→
   890→    # 初始化日志
   891→    setup_logging()
   892→
   893→    logger.info("=" * 60)

   897→    if args.mcp:
   898→        logger.info(f"MCP: http://0.0.0.0:{args.mcp_port}")
   899→    logger.info("=" * 60)
   900→
   901→    # 如果需要，启动 MCP 服务器
   902→    if args.mcp:
   903→        start_mcp_server_background(port=args.mcp_port)
   904→
   905→    # 启动 GUI
   906→    app = create_app()
   907→    app.launch(
   908→        server_name=args.host,
   909→        server_port=args.port,
   910→        share=False,
   911→        inbrowser=True,
   912→    )

   914→
   915→if __name__ == "__main__":
   916→    main()
